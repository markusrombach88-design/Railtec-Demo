"""RailTecSol 0.3: searchable technical manuals, drop-in Streamlit module."""
from __future__ import annotations
import hashlib
import io
import os
import re
from pathlib import Path
from collections import Counter
import math
import streamlit as st
from pypdf import PdfReader
from docx import Document

BASE = Path(__file__).resolve().parent
MANUALS = BASE / 'manuals'
EXTS = {'.pdf', '.docx', '.txt'}

def extract(data: bytes, filename: str):
    suffix = Path(filename).suffix.lower()
    if suffix == '.pdf':
        reader = PdfReader(io.BytesIO(data))
        return [(i + 1, page.extract_text() or '') for i, page in enumerate(reader.pages)]
    if suffix == '.docx':
        doc = Document(io.BytesIO(data))
        # DOCX does not have reliable fixed page numbers.
        blocks = [p.text for p in doc.paragraphs if p.text.strip()]
        for table in doc.tables:
            for row in table.rows:
                blocks.append(' | '.join(cell.text for cell in row.cells))
        return [(None, '\n'.join(blocks))]
    if suffix == '.txt':
        return [(None, data.decode('utf-8-sig', errors='replace'))]
    raise ValueError('Nicht unterstütztes Dateiformat')

def chunks(data: bytes, name: str, origin: str):
    out = []
    for page, content in extract(data, name):
        content = re.sub(r'[ \t]+', ' ', content).strip()
        if not content:
            continue
        # Keep a small overlap to preserve context between passages.
        for start in range(0, len(content), 1250):
            snippet = content[start:start+1500]
            if len(snippet.strip()) < 25:
                continue
            out.append({'name': name, 'page': page, 'text': snippet, 'origin': origin})
            if start + 1500 >= len(content):
                break
    return out

@st.cache_data(show_spinner=False, max_entries=12)
def parse_cached(data: bytes, name: str, origin: str):
    return chunks(data, name, origin)

def tokenize(s: str):
    return re.findall(r'[a-zäöüß0-9]+', s.lower())

def rank(query: str, passages: list, limit: int = 6):
    terms = tokenize(query)
    if not terms or not passages:
        return []
    docs = [Counter(tokenize(p['text'])) for p in passages]
    lengths = [sum(d.values()) for d in docs]
    avg = sum(lengths) / len(lengths) or 1
    df = Counter(t for d in docs for t in d)
    scores = []
    for i, counts in enumerate(docs):
        score = 0.0
        for t in set(terms):
            freq = counts[t]
            if not freq:
                continue
            idf = math.log(1 + (len(docs) - df[t] + .5) / (df[t] + .5))
            score += idf * freq * 2.2 / (freq + 1.2 * (.25 + .75 * lengths[i] / avg))
        if score > 0:
            scores.append((score, i))
    scores.sort(reverse=True)
    return [passages[i] for _, i in scores[:limit]]

def api_key():
    try:
        return st.secrets.get('OPENAI_API_KEY', '') or os.getenv('OPENAI_API_KEY', '')
    except Exception:
        return os.getenv('OPENAI_API_KEY', '')

def answer(question: str, passages: list):
    from openai import OpenAI
    key = api_key()
    if not key:
        return None
    context = '\n\n'.join(
        f"[Quelle {i}] Datei: {p['name']} | Seite: {p['page'] if p['page'] is not None else 'nicht verfügbar'}\n{p['text']}"
        for i, p in enumerate(passages, 1)
    )
    client = OpenAI(api_key=key)
    response = client.chat.completions.create(
        model='gpt-4.1-mini',
        temperature=0.1,
        messages=[
            {'role': 'system', 'content': 'Du bist ein technischer Assistent für Schienenfahrzeug-Instandhaltung. Antworte ausschließlich anhand der bereitgestellten Auszüge. Zitiere jede technische Aussage mit [Quelle N] und, wenn vorhanden, Seitenzahl. Wenn die Quellen keine ausreichende Antwort liefern, sage das klar. Erfinde keine Grenzwerte, Prüfverfahren oder Sicherheitsfreigaben. Bei sicherheitskritischen Tätigkeiten gelten die freigegebenen betrieblichen Anweisungen.'},
            {'role': 'user', 'content': f'Quellen:\n{context}\n\nFrage: {question}'}
        ]
    )
    return response.choices[0].message.content

def render_knowledge_base():
    st.header('📚 Technische Wissensdatenbank')
    st.caption('Dauerhafte Manuals im GitHub-Ordner manuals/ · zusätzliche Uploads nur für diese Sitzung')
    MANUALS.mkdir(exist_ok=True)
    files = sorted(p for p in MANUALS.iterdir() if p.is_file() and p.suffix.lower() in EXTS)
    st.subheader('Dauerhafte Manuals')
    if files:
        for p in files:
            st.write('📄', p.name)
    else:
        st.info('Noch keine Manuals hinterlegt. Lade Dateien über GitHub in den Ordner manuals/ hoch.')
    st.subheader('Zusätzliche Dokumente')
    uploads = st.file_uploader('PDF, DOCX oder TXT auswählen', type=['pdf','docx','txt'], accept_multiple_files=True, key='railtec_uploads')
    passages = []
    errors = []
    with st.spinner('Dokumente werden eingelesen ...'):
        for p in files:
            try:
                if p.stat().st_size > 20 * 1024 * 1024:
                    errors.append(f'{p.name}: größer als 20 MB')
                    continue
                passages.extend(parse_cached(p.read_bytes(), p.name, 'manual'))
            except Exception as exc:
                errors.append(f'{p.name}: {type(exc).__name__}')
        for f in uploads or []:
            try:
                if f.size > 20 * 1024 * 1024:
                    errors.append(f'{f.name}: größer als 20 MB')
                    continue
                passages.extend(parse_cached(f.getvalue(), f.name, 'upload'))
            except Exception as exc:
                errors.append(f'{f.name}: {type(exc).__name__}')
    for error in errors:
        st.warning(f'Dokument nicht lesbar: {error}')
    st.caption(f'{len(files)} dauerhafte Datei(en) · {len(uploads or [])} temporäre Datei(en) · {len(passages)} durchsuchbare Textabschnitte')
    st.info('Eingescannte PDFs ohne Textschicht benötigen vorher eine Texterkennung (OCR). Bei Word-Dateien sind keine verlässlichen Seitenzahlen verfügbar.')
    question = st.text_input('Frage an deine Dokumente', placeholder='Welche Prüfschritte sind für die KE-GP-A-Bremse beschrieben?', key='railtec_question')
    if st.button('In Dokumenten suchen', type='primary', key='railtec_search'):
        if not question.strip():
            st.warning('Bitte zuerst eine Frage eingeben.')
        elif not passages:
            st.warning('Keine lesbaren Dokumente vorhanden.')
        else:
            matches = rank(question, passages)
            if not matches:
                st.warning('Keine passende Textstelle gefunden. Versuche andere Suchbegriffe.')
            else:
                with st.spinner('Antwort wird erstellt ...'):
                    try:
                        result = answer(question, matches)
                        if result:
                            st.markdown(result)
                        else:
                            st.info('Kein OPENAI_API_KEY eingerichtet. Die gefundenen Fundstellen stehen unten.')
                    except Exception as exc:
                        st.error(f'KI-Antwort nicht verfügbar ({type(exc).__name__}). Fundstellen sind weiterhin sichtbar.')
                st.subheader('Fundstellen zum Nachprüfen')
                for i, p in enumerate(matches, 1):
                    page = f'Seite {p["page"]}' if p['page'] is not None else 'ohne feste Seitenzahl'
                    with st.expander(f'Quelle {i}: {p["name"]} · {page}', expanded=i==1):
                        st.text(p['text'])
